# Core runtime dependencies for the Streamlit app.
# Install with:  pip install -r requirements.txt
#
#   streamlit        -> web UI (python -m streamlit run app.py)
#   requests         -> HTTP client for SoundLoaders / SpotiDown (providers/*)
#   beautifulsoup4   -> HTML parsing of provider responses (providers/parsing.py)
#   lxml             -> fast HTML parser used by BeautifulSoup (optional but
#                       recommended; code falls back to html.parser if missing)
#   jinja2           -> bundled/streamlit ecosystem template dependency; kept
#                       explicit so environment versions stay consistent
#   playwright       -> drives a REAL Chrome over CDP for spotidown.co
#                       (providers/spotidown_co.py). Only the Python package is
#                       needed; we never install Chromium browsers because the
#                       provider attaches to the Chrome already on the machine.
#   mutagen            -> writes Spotify ID3 tags + embedded cover art onto the
#                       downloaded MP3 (providers/tags.py). Tags only: the audio
#                       stream is never re-encoded.

streamlit>=1.40
requests>=2.31

playwright>=1.44
